Google Apps Script 코드
Drive 자동 업로드 + 뉴스 발송
아래 코드를 script.google.com에서 새 프로젝트로 만들고, 웹앱 배포 후 URL을 설정 탭에 입력하세요.
function doPost(e) {
  const name = 'FamilyOffice_Backup';
  const folder = DriveApp.getFoldersByName(name).hasNext()
    ? DriveApp.getFoldersByName(name).next()
    : DriveApp.createFolder(name);
  const d = JSON.parse(e.postData.contents);
  const blob = Utilities.newBlob(
    Utilities.base64Decode(d.base64),
    d.mimeType, d.fileName);
  folder.createFile(blob);
  return ContentService
    .createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}
